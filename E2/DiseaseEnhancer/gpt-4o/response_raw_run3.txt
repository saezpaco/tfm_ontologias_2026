### Classes
:CRM rdf:type owl:Class ;
    skos:prefLabel "Cis-regulatory module" ;
    skos:definition "Cis-regulatory module located in Homo sapiens chr_name between start_position and end_position" .

### Object Properties
:hasTargetGene rdf:type owl:ObjectProperty ;
    rdfs:domain :CRM ;
    rdfs:range :Gene ;
    skos:prefLabel "has target gene" ;
    skos:definition "Association between cis-regulatory module and target gene" .

:associatedWithDisease rdf:type owl:ObjectProperty ;
    rdfs:domain :CRM ;
    rdfs:range :Disease ;
    skos:prefLabel "associated with disease" ;
    skos:definition "Association between cis-regulatory module and disease" .

:interactsWithTF rdf:type owl:ObjectProperty ;
    rdfs:domain :CRM ;
    rdfs:range :TranscriptionFactor ;
    skos:prefLabel "interacts with transcription factor" ;
    skos:definition "Association between cis-regulatory module and transcription factor" .

:hasGenomicCoordinates rdf:type owl:DatatypeProperty ;

:hasScore rdf:type owl:DatatypeProperty ;

    rdfs:range xsd:float ;
    skos:prefLabel "has score" ;
    skos:definition "Score of the cis-regulatory module" .

:hasMinimumRatio rdf:type owl:DatatypeProperty ;
    rdfs:domain :CRM ;

    skos:prefLabel "has minimum ratio" ;
    skos:definition "Minimum ratio of the cis-regulatory module" .
