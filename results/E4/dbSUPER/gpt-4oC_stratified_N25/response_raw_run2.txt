################ DATA DESCRIPTION (PlanSage) ################
**Foundational Prefix:**
@prefix base: <https://base_ontology.com#> .
@prefix rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#> .
@prefix rdfs: <http://www.w3.org/2000/01/rdf-schema#> .
@prefix skos: <http://www.w3.org/2004/02/skos/core#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .
@prefix owl: <http://www.w3.org/2002/07/owl#> .

**Data Description:**
1. cell_name: text - name of the cell line or biosample - string
2. chrom: text - chromosome identifier - string
3. gene_symbol: text - symbol of the target gene - string
4. se_id: text - identifier for the super-enhancer - string
5. rank: Numerical - rank of the super-enhancer - 45.0 to 1247.0
6. start: Numerical - start coordinate of the genomic region - 49148.0 to 187454355.0
7. stop: Numerical - stop coordinate of the genomic region - 60599.0 to 187468841.0

**classes:**
1. GeneticElement
2. GenomicCoordinate
3. CellLine
4. Gene
5. SuperEnhancer

**subclasses:**
1. Enhancer: subclass of -> GeneticElement

**Object Properties:**
1. hasGenomicCoordinate: GeneticElement - GenomicCoordinate
2. isLocatedIn: GenomicCoordinate - CellLine
3. targetsGene: GeneticElement - Gene
4. hasSuperEnhancer: CellLine - SuperEnhancer

**Data Type Properties:**
1. cellName: CellLine - xsd:string
2. chrom: GenomicCoordinate - xsd:string
3. geneSymbol: Gene - xsd:string
4. seId: SuperEnhancer - xsd:string
5. rank: SuperEnhancer - xsd:float
6. start: GenomicCoordinate - xsd:float
7. stop: GenomicCoordinate - xsd:float

################ ONTOLOGY (OntoBuilder) #####################

base:cellName a owl:DatatypeProperty ;
    rdfs:domain base:CellLine ;
    rdfs:range xsd:string ;
    rdfs:label "cell name" ;
    skos:definition "name of the cell line or biosample" .

base:chrom a owl:DatatypeProperty ;
    rdfs:domain base:GenomicCoordinate ;
    rdfs:range xsd:string ;
    rdfs:label "chromosome" ;
    skos:definition "chromosome identifier" .

base:geneSymbol a owl:DatatypeProperty ;
    rdfs:domain base:Gene ;
    rdfs:range xsd:string ;
    rdfs:label "gene symbol" ;
    skos:definition "symbol of the target gene" .

base:seId a owl:DatatypeProperty ;
    rdfs:domain base:SuperEnhancer ;
    rdfs:range xsd:string ;
    rdfs:label "super-enhancer ID" ;
    skos:definition "identifier for the super-enhancer" .

base:rank a owl:DatatypeProperty ;
    rdfs:domain base:SuperEnhancer ;
    rdfs:range xsd:float ;
    rdfs:label "rank" ;
    skos:definition "rank of the super-enhancer" .

base:start a owl:DatatypeProperty ;
    rdfs:domain base:GenomicCoordinate ;
    rdfs:range xsd:float ;
    rdfs:label "start coordinate" ;
    skos:definition "start coordinate of the genomic region" .

base:stop a owl:DatatypeProperty ;
    rdfs:domain base:GenomicCoordinate ;
    rdfs:range xsd:float ;
    rdfs:label "stop coordinate" ;
    skos:definition "stop coordinate of the genomic region" .

base:hasGenomicCoordinate a owl:ObjectProperty ;
    rdfs:domain base:GeneticElement ;
    rdfs:range base:GenomicCoordinate ;
    rdfs:label "has genomic coordinate" ;
    skos:definition "links a genetic element to its genomic coordinate" .

base:isLocatedIn a owl:ObjectProperty ;
    rdfs:domain base:GenomicCoordinate ;
    rdfs:range base:CellLine ;
    rdfs:label "is located in" ;
    skos:definition "links a genomic coordinate to the cell line it is located in" .

base:targetsGene a owl:ObjectProperty ;
    rdfs:domain base:GeneticElement ;
    rdfs:range base:Gene ;
    rdfs:label "targets gene" ;
    skos:definition "links a genetic element to the gene it targets" .

base:hasSuperEnhancer a owl:ObjectProperty ;
    rdfs:domain base:CellLine ;
    rdfs:range base:SuperEnhancer ;
    rdfs:label "has super-enhancer" ;
    skos:definition "links a cell line to its super-enhancer" .

base:GeneticElement a owl:Class ;
    rdfs:label "Genetic Element" ;
    skos:definition "A genetic element" ;
    rdfs:subClassOf [
        a owl:Restriction ;
        owl:onProperty base:hasGenomicCoordinate ;
        owl:someValuesFrom base:GenomicCoordinate
    ], [
        a owl:Restriction ;
        owl:onProperty base:targetsGene ;
        owl:someValuesFrom base:Gene
    ] .

base:GenomicCoordinate a owl:Class ;
    rdfs:label "Genomic Coordinate" ;
    skos:definition "A genomic coordinate" ;
    rdfs:subClassOf [
        a owl:Restriction ;
        owl:onProperty base:isLocatedIn ;
        owl:someValuesFrom base:CellLine
    ] .

base:CellLine a owl:Class ;
    rdfs:label "Cell Line" ;
    skos:definition "A cell line or biosample" ;
    rdfs:subClassOf [
        a owl:Restriction ;
        owl:onProperty base:hasSuperEnhancer ;
        owl:someValuesFrom base:SuperEnhancer
    ] .

base:Gene a owl:Class ;
    rdfs:label "Gene" ;
    skos:definition "A gene" .

base:SuperEnhancer a owl:Class ;
    rdfs:label "Super Enhancer" ;
    skos:definition "A super-enhancer" ;
    rdfs:subClassOf base:Enhancer .

base:Enhancer a owl:Class ;
    rdfs:label "Enhancer" ;
    skos:definition "An enhancer" ;
    rdfs:subClassOf base:GeneticElement .
