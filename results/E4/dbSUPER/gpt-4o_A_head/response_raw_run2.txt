################ DATA DESCRIPTION (PlanSage) ################
**Foundational Prefix:**
@prefix base: <https://base_ontology.com#> .
@prefix rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#> .
@prefix rdfs: <http://www.w3.org/2000/01/rdf-schema#> .
@prefix skos: <http://www.w3.org/2004/02/skos/core#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .
@prefix owl: <http://www.w3.org/2002/07/owl#> .

**Data Description:**
1. cell_name: categorical - name of the cell line or biosample - single value
2. chrom: text - chromosome identifier - multiple values
3. gene_symbol: text - symbol of the gene - multiple values
4. se_id: text - identifier for the super-enhancer - single value
5. rank: Numerical - rank of the super-enhancer - range: 1 to 25
6. start: Numerical - start position of the genomic coordinate - range: 12327760 to 218548183
7. stop: Numerical - stop position of the genomic coordinate - range: 12522419 to 218895526

**classes:**
1. GeneticElement
2. GenomicCoordinate
3. CellLine
4. Gene
5. SuperEnhancer

**subclasses:**
1. Enhancer: subclass of -> GeneticElement
2. SuperEnhancer: subclass of -> Enhancer

**Object Properties:**
1. hasGenomicCoordinate: GeneticElement - GenomicCoordinate
2. isLocatedIn: GenomicCoordinate - CellLine
3. targetsGene: GeneticElement - Gene
4. hasSuperEnhancerID: SuperEnhancer - xsd:string

**Data Type Properties:**
1. cellName: CellLine - xsd:string
2. chromosome: GenomicCoordinate - xsd:string
3. geneSymbol: Gene - xsd:string
4. rank: SuperEnhancer - xsd:integer
5. start: GenomicCoordinate - xsd:integer
6. stop: GenomicCoordinate - xsd:integer

base:cellName a owl:DatatypeProperty ;
    rdfs:domain base:CellLine ;
    rdfs:range xsd:string ;
    rdfs:label "cell name" ;
    skos:definition "Name of the cell line or biosample" .

    rdfs:domain base:GenomicCoordinate ;
    rdfs:range xsd:string ;
    rdfs:label "chromosome" ;
    skos:definition "Chromosome identifier" .

base:geneSymbol a owl:DatatypeProperty ;
    rdfs:domain base:Gene ;
    rdfs:range xsd:string ;
    rdfs:label "gene symbol" ;
    skos:definition "Symbol of the gene" .

base:rank a owl:DatatypeProperty ;
    rdfs:domain base:SuperEnhancer ;
    rdfs:range xsd:integer ;
    rdfs:label "rank" ;
    skos:definition "Rank of the super-enhancer" .

base:start a owl:DatatypeProperty ;
    rdfs:domain base:GenomicCoordinate ;
    rdfs:range xsd:integer ;
    rdfs:label "start" ;
    skos:definition "Start position of the genomic coordinate" .

base:stop a owl:DatatypeProperty ;
    rdfs:domain base:GenomicCoordinate ;
    rdfs:range xsd:integer ;
    rdfs:label "stop" ;
    skos:definition "Stop position of the genomic coordinate" .

base:hasGenomicCoordinate a owl:ObjectProperty ;
    rdfs:domain base:GeneticElement ;
    rdfs:range base:GenomicCoordinate ;

    rdfs:domain base:GenomicCoordinate ;

base:targetsGene a owl:ObjectProperty ;
    rdfs:domain base:GeneticElement ;
    rdfs:range base:Gene ;
    rdfs:label "targets gene" ;
    skos:definition "Indicates the gene targeted by a genetic element" .

base:hasSuperEnhancerID a owl:ObjectProperty ;
    rdfs:domain base:SuperEnhancer ;
    rdfs:range xsd:string ;
    rdfs:label "has super-enhancer ID" ;
    skos:definition "Identifier for the super-enhancer" .

base:GeneticElement a owl:Class ;
    rdfs:label "Genetic Element" ;
    skos:definition "A genetic element" .

base:GenomicCoordinate a owl:Class ;
    rdfs:label "Genomic Coordinate" ;
    skos:definition "A genomic coordinate" .

base:CellLine a owl:Class ;
    rdfs:label "Cell Line" ;
    skos:definition "A cell line or biosample" .

base:Gene a owl:Class ;
    rdfs:label "Gene" ;
    skos:definition "A gene" .

base:Enhancer a owl:Class ;
    rdfs:subClassOf base:GeneticElement ;
    rdfs:label "Enhancer" ;
    skos:definition "An enhancer, a subclass of genetic element" .

base:SuperEnhancer a owl:Class ;
    rdfs:subClassOf base:Enhancer ;
    rdfs:label "Super Enhancer" ;
    skos:definition "A super-enhancer, a subclass of enhancer" .

base:SuperEnhancer a owl:Class ;
    rdfs:subClassOf [
        a owl:Restriction ;
        owl:onProperty base:hasSuperEnhancerID ;
        owl:allValuesFrom xsd:string
    ] .

base:GeneticElement a owl:Class ;
    rdfs:subClassOf [
        a owl:Restriction ;
        owl:onProperty base:hasGenomicCoordinate ;
        owl:allValuesFrom base:GenomicCoordinate
    ] .

base:GenomicCoordinate a owl:Class ;
    rdfs:subClassOf [
        a owl:Restriction ;
        owl:onProperty base:isLocatedIn ;
        owl:allValuesFrom base:CellLine
    ] .

base:GeneticElement a owl:Class ;
    rdfs:subClassOf [
        a owl:Restriction ;
        owl:onProperty base:targetsGene ;
        owl:allValuesFrom base:Gene
    ] .

<#TriplesMapCellLine> a rr:TriplesMap;
    rml:logicalSource [
        rml:source "dbSUPER.csv";
        rml:referenceFormulation ql:CSV
    ];
    rr:subjectMap [
        rr:template "https://base_ontology.com#CellLine/{cell_name}";
    ];
    rr:predicateObjectMap [
        rr:predicate base:cellName;
        rr:objectMap [
            rml:reference "cell_name"
        ]
    ].
