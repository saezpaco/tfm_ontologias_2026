    rdfs:label "HACER Ontology" ;

### Classes ###
:CRM a owl:Class ;
    skos:prefLabel "CRM" ;
    skos:definition "Cis-regulatory module located in Homo sapiens chr_name between start_position and end_position" .

:Gene a owl:Class ;
    skos:prefLabel "Gene" ;
    skos:definition "A gene that is a target of a cis-regulatory module" .

:TranscriptionFactor a owl:Class ;
    skos:prefLabel "Transcription Factor" ;
    skos:definition "A transcription factor that interacts with a cis-regulatory module" .

:Disease a owl:Class ;
    skos:prefLabel "Disease" ;
    skos:definition "A disease associated with a cis-regulatory module" .

:CellLine a owl:Class ;
    skos:prefLabel "Cell Line" ;
    skos:definition "A cell line in which the cis-regulatory module is active" .

    rdfs:domain :CRM ;
