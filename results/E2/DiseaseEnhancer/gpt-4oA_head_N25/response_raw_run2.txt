```turtle
@prefix : <http://example.org/diseaseenhancer#> .
@prefix rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#> .
@prefix rdfs: <http://www.w3.org/2000/01/rdf-schema#> .
@prefix owl: <http://www.w3.org/2002/07/owl#> .
@prefix skos: <http://www.w3.org/2004/02/skos/core#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .
@prefix obo: <http://purl.obolibrary.org/obo/> .

:CRM a owl:Class ;
    skos:prefLabel "CRM" ;
    skos:definition "Cis-regulatory module located in Homo sapiens chr_name between start_position and end_position" .

:Gene a owl:Class ;
    skos:prefLabel "Gene" .

:Disease a owl:Class ;
    skos:prefLabel "Disease" .

:TranscriptionFactor a owl:Class ;
    skos:prefLabel "Transcription Factor" .

:hasGenomicLocation a owl:ObjectProperty ;
    rdfs:domain :CRM ;
    rdfs:range :GenomicLocation ;
    skos:prefLabel "has genomic location" .

:GenomicLocation a owl:Class ;
    skos:prefLabel "Genomic Location" .

:hasTargetGene a owl:ObjectProperty ;
    rdfs:domain :CRM ;
    rdfs:range :Gene ;
    skos:prefLabel "has target gene" ;
    skos:definition "Association between cis-regulatory module and gene" ;
    skos:exactMatch :bgw!crm_ID--hgncsymbol!9606/gene_symbol .

:associatedWithDisease a owl:ObjectProperty ;
    rdfs:domain :CRM ;
    rdfs:range :Disease ;
    skos:prefLabel "associated with disease" ;
    skos:definition "Association between cis-regulatory module and disease" ;
    skos:exactMatch :bgw!crm_ID--doid!doid .

:interactsWithTranscriptionFactor a owl:ObjectProperty ;
    rdfs:domain :CRM ;
    rdfs:range :TranscriptionFactor ;
    skos:prefLabel "interacts with transcription factor" ;
    skos:definition "Association between cis-regulatory module and transcription factor" ;
    skos:exactMatch :bgw!crm_ID--uniprot!uniprot_ID .

:chr_name a owl:DatatypeProperty ;
    rdfs:domain :GenomicLocation ;
    rdfs:range xsd:string ;
    skos:prefLabel "chr_name" .

:start_position a owl:DatatypeProperty ;
    rdfs:domain :GenomicLocation ;
    rdfs:range xsd:integer ;
    skos:prefLabel "start position" .

:end_position a owl:DatatypeProperty ;
    rdfs:domain :GenomicLocation ;
    rdfs:range xsd:integer ;
    skos:prefLabel "end position" .

:database_name a owl:DatatypeProperty ;
    rdfs:domain :CRM ;
    rdfs:range xsd:string ;
    skos:prefLabel "Database" .

:GRCh38 a owl:NamedIndividual ;
    rdf:type :GenomicLocation ;
    skos:prefLabel "GRCh38" .
```
