# Ontology definition
<http://example.org/DiseaseEnhancerOntology> a owl:Ontology ;
    dc:title "Disease Enhancer Ontology" ;
    dc:description "An ontology for enhancers associated with diseases" ;
    dc:source <http://biocc.hrbmu.edu.cn/DiseaseEnhancer/> .

# Classes
hcrm:CisRegulatoryModule a owl:Class ;
    rdfs:label "Cis-Regulatory Module" ;
    skos:exactMatch crm2gene:CisRegulatoryModule .

hcrm:Enhancer a owl:Class ;
    rdfs:subClassOf hcrm:CisRegulatoryModule ;
    rdfs:label "Enhancer" ;
    skos:exactMatch crm2gene:Enhancer .

hcrm:Disease a owl:Class ;
    rdfs:label "Disease" ;
    skos:exactMatch obo:DOID_4 .

hcrm:Gene a owl:Class ;
    rdfs:label "Gene" ;
    skos:exactMatch biolink:Gene .

hcrm:TranscriptionFactor a owl:Class ;
    rdfs:label "Transcription Factor" ;
    skos:exactMatch biolink:TranscriptionFactor .

# Properties
hcrm:hasGenomicCoordinate a owl:ObjectProperty ;
    rdfs:label "has genomic coordinate" ;
    skos:closeMatch crm2gene:hasGenomicCoordinate .

hcrm:hasTargetGene a owl:ObjectProperty ;
    rdfs:label "has target gene" ;
    skos:exactMatch crm2gene:hasTargetGene .

hcrm:associatedWithDisease a owl:ObjectProperty ;
    rdfs:label "associated with disease" ;
    skos:exactMatch biolink:associatedWithDisease .

hcrm:interactsWithTF a owl:ObjectProperty ;
    rdfs:label "interacts with transcription factor" ;
    skos:exactMatch biolink:interactsWith .

hcrm:hasExperimentalEvidence a owl:DatatypeProperty ;
    rdfs:label "has experimental evidence" ;
    skos:closeMatch crm2gene:hasEvidence .

hcrm:hasPMID a owl:DatatypeProperty ;
    rdfs:label "has PMID" ;
    skos:exactMatch pubmed:PMID .

hcrm:hasMutation a owl:DatatypeProperty ;
    rdfs:label "has mutation" ;
    skos:exactMatch obo:SO_0001060 .
