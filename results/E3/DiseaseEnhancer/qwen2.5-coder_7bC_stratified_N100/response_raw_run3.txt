# Define classes and properties based on the data
crm:TranscriptionFactor a owl:Class ;
    rdfs:label "Transcription Factor" ;
    skos:exactMatch obo:NCIT_C17207 .

crm:Disease a owl:Class ;
    rdfs:label "Disease" ;
    skos:closeMatch hcrm:Disease .

crm:Biosample a owl:Class ;
    rdfs:label "Biosample" ;
    skos:exactMatch obo:SO_0001217 .

crm:Gene a owl:Class ;
    rdfs:label "Gene" ;
    skos:closeMatch hgene:Gene .

crm:CisRegulatoryElement a owl:Class ;
    rdfs:label "Cis-Regulatory Element" ;
    skos:exactMatch obo:SO_0001217 .

crm:Chromosome a owl:Class ;
    rdfs:label "Chromosome" ;
    skos:closeMatch assembly:Chromosome .

crm:NucleotideSequence a owl:Class ;
    rdfs:label "Nucleotide Sequence" ;
    skos:exactMatch nuccore:NucleotideSequence .

crm:PubMedArticle a owl:Class ;
    rdfs:label "PubMed Article" ;
    skos:closeMatch pubmed:Article .

crm:GeneSymbol a owl:Class ;
    rdfs:label "Gene Symbol" ;
    skos:closeMatch hgene:GeneSymbol .

    rdfs:label "Disease PubMed ID" ;
    rdfs:domain crm:Disease ;
    rdfs:range crm:PubMedArticle ;
    skos:exactMatch id_pubmed:.

crm:BiosampleName a owl:Property ;
    rdfs:label "Biosample Name" ;
    rdfs:domain crm:Biosample ;
    rdfs:range xsd:string ;
    skos:closeMatch obo:SO_0001217 .

    rdfs:label "Gene Symbol Target Genes" ;
    rdfs:domain crm:TranscriptionFactor ;
    rdfs:range crm:Gene ;
    skos:closeMatch hgene:GeneSybmol .

crm:CisRegulatoryElementOrigEnd a owl:Property ;

    rdfs:domain crm:CisRegulatoryElement ;
    rdfs:range xsd:string ;
    skos:exactMatch obo:SO_0001217 .

crm:CisRegulatoryElementCurrentEnd a owl:Property ;
    rdfs:label "Cis-Regulatory Element Current End" ;
    rdfs:domain crm:CisRegulatoryElement ;
    rdfs:range xsd:string ;
    skos:exactMatch obo:SO_0001217 .

crm:CisRegulatoryElementChromosome a owl:Property ;
    rdfs:label "Cis-Regulatory Element Chromosome" ;
    rdfs:domain crm:CisRegulatoryElement ;
    rdfs:range crm:Chromosome ;
    skos:closeMatch assembly:Chromosome .

crm:CisRegulatoryElementNucleotideSequence a owl:Property ;
    rdfs:label "Cis-Regulatory Element Nucleotide Sequence" ;
    rdfs:domain crm:CisRegulatoryElement ;
    rdfs:range crm:NucleotideSequence ;
    skos:exactMatch nuccore:NucleotideSequence .

crm:TranscriptionFactor1 a crm:TranscriptionFactor ;
