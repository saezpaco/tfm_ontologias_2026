```turtle
@prefix rdf:     <http://www.w3.org/1999/02/22-rdf-syntax-ns#> .
@prefix rdfs:    <http://www.w3.org/2000/01/rdf-schema#> .
@prefix owl:     <http://www.w3.org/2002/07/owl#> .
@prefix xsd:     <http://www.w3.org/2001/XMLSchema#> .
@prefix skos:    <http://www.w3.org/2004/02/skos/core#> .
@prefix dc:      <http://purl.org/dc/terms/> .
@prefix obo:     <http://purl.obolibrary.org/obo/> .
@prefix biolink: <https://w3id.org/biolink/vocab/> .
@prefix schema:  <http://schema.org/> .
@prefix sio:     <http://semanticscience.org/resource/> .
@prefix crm2pgene: <http://rdf.biogateway.eu/crm2pgene/> .
@prefix crm2tfac: <http://rdf.biogateway.eu/crm2tfac/> .
@prefix dbsuper: <https://asntech.org/dbsuper/> .
@prefix pubmed:  <https://pubmed.ncbi.nlm.nih.gov/> .

# Ontology definition
<http://example.org/dbSUPER_ontology> a owl:Ontology ;
    rdfs:label "dbSUPER Ontology" ;
    dc:description "Ontology for representing super-enhancers from the dbSUPER database." .

# Classes
dbsuper:SuperEnhancer a owl:Class ;
    rdfs:label "Super Enhancer" ;
    rdfs:subClassOf biolink:Enhancer ;
    skos:exactMatch crm2pgene:SuperEnhancer .

dbsuper:Gene a owl:Class ;
    rdfs:label "Gene" ;
    skos:exactMatch biolink:Gene .

dbsuper:CellLine a owl:Class ;
    rdfs:label "Cell Line" ;
    skos:exactMatch biolink:CellLine .

# Properties
dbsuper:hasGenomicCoordinate a owl:ObjectProperty ;
    rdfs:label "has genomic coordinate" ;
    rdfs:domain dbsuper:SuperEnhancer ;
    rdfs:range sio:SIO_000614 ; # Genomic coordinate
    skos:closeMatch crm2pgene:hasGenomicCoordinate .

dbsuper:targetsGene a owl:ObjectProperty ;
    rdfs:label "targets gene" ;
    rdfs:domain dbsuper:SuperEnhancer ;
    rdfs:range dbsuper:Gene ;
    skos:exactMatch crm2pgene:targetsGene .

dbsuper:activeInCellLine a owl:ObjectProperty ;
    rdfs:label "active in cell line" ;
    rdfs:domain dbsuper:SuperEnhancer ;
    rdfs:range dbsuper:CellLine ;
    skos:exactMatch crm2tfac:activeInCellLine .

dbsuper:hasExperimentalEvidence a owl:DatatypeProperty ;
    rdfs:label "has experimental evidence" ;
    rdfs:domain dbsuper:SuperEnhancer ;
    rdfs:range xsd:string ;
    skos:closeMatch crm2tfac:hasExperimentalEvidence .

dbsuper:hasPMID a owl:DatatypeProperty ;
    rdfs:label "has PMID" ;
    rdfs:domain dbsuper:SuperEnhancer ;
    rdfs:range pubmed: ;
    skos:exactMatch crm2pgene:hasPMID .

# Example individuals
dbsuper:CRMHS00000005972 a dbsuper:SuperEnhancer ;

        a sio:SIO_000614 ;
```
