@prefix skos:    <http://www.w3.org/2004/02/skos/core#> .
@prefix dc:      <http://purl.org/dc/terms/> .

@prefix crm2pgene: <http://example.org/ontologies/crm2pgene#> .
@prefix hgene:   <http://example.org/ontologies/hgene#> .
@prefix assembly: <http://example.org/ontologies/assembly#> .
@prefix nuccore: <http://example.org/ontologies/nuccore#> .

# Ontology definition
<http://asntech.org/dbsuper/ontology> a owl:Ontology ;
    dc:title "dbSUPER Ontology" ;
    dc:description "Ontology for representing super-enhancers from the dbSUPER database." ;
    dc:source <https://asntech.org/dbsuper/> .

# Classes
dbsuper:SuperEnhancer a owl:Class ;
    rdfs:label "Super Enhancer" ;
    skos:exactMatch hcrm:Enhancer .

dbsuper:Gene a owl:Class ;
    rdfs:label "Gene" ;
    skos:exactMatch hgene:Gene .

dbsuper:CellLine a owl:Class ;
    rdfs:label "Cell Line" ;
    skos:exactMatch crm2pgene:CellLine .

# Properties
dbsuper:hasGenomicCoordinate a owl:ObjectProperty ;
    rdfs:label "has genomic coordinate" ;
    skos:exactMatch assembly:GenomicCoordinate .

dbsuper:hasStart a owl:DatatypeProperty ;
    rdfs:label "has start position" ;
    rdfs:domain dbsuper:SuperEnhancer ;
    rdfs:range xsd:integer ;
    skos:exactMatch nuccore:Start .

dbsuper:hasEnd a owl:DatatypeProperty ;
    rdfs:label "has end position" ;
    rdfs:domain dbsuper:SuperEnhancer ;
    rdfs:range xsd:integer ;
    skos:exactMatch nuccore:End .

dbsuper:hasChromosome a owl:DatatypeProperty ;
    rdfs:label "has chromosome" ;
    rdfs:domain dbsuper:SuperEnhancer ;
    rdfs:range xsd:string ;
    skos:exactMatch assembly:Chromosome .

dbsuper:targetsGene a owl:ObjectProperty ;
    rdfs:label "targets gene" ;
    rdfs:domain dbsuper:SuperEnhancer ;
    rdfs:range dbsuper:Gene ;
    skos:exactMatch crm2pgene:targetsGene .

dbsuper:activeInCellLine a owl:ObjectProperty ;
    rdfs:label "active in cell line" ;
    rdfs:domain dbsuper:SuperEnhancer ;
    rdfs:range dbsuper:CellLine ;
    skos:exactMatch crm2pgene:activeInCellLine .

dbsuper:hasRank a owl:DatatypeProperty ;
    rdfs:label "has rank" ;
    rdfs:domain dbsuper:SuperEnhancer ;
    rdfs:range xsd:integer .

# Example individuals
dbsuper:SE_00001 a dbsuper:SuperEnhancer ;
    dbsuper:hasChromosome "chr1" ;
    dbsuper:hasStart 120485363 ;
    dbsuper:hasEnd 120615071 ;
    dbsuper:targetsGene dbsuper:NOTCH2 ;
    dbsuper:activeInCellLine dbsuper:AdiposeNuclei ;
    dbsuper:hasRank 1 .

dbsuper:NOTCH2 a dbsuper:Gene ;
    rdfs:label "NOTCH2" ;
    skos:exactMatch id_ncbigene:4851 .

dbsuper:AdiposeNuclei a dbsuper:CellLine ;
    rdfs:label "Adipose Nuclei" .
