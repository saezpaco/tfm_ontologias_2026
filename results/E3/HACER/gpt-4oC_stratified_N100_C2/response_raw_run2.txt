# Ontology definition
<http://bioinfo.vanderbilt.edu/AE/HACER/> rdf:type owl:Ontology ;
    dc:title "Human Active Cis-regulatory Elements database" ;
    dc:description "Ontology schema for the HACER database, representing cis-regulatory modules and their properties." .

# Classes
hcrm:CisRegulatoryModule rdf:type owl:Class ;
    rdfs:label "Cis-Regulatory Module" ;
    skos:exactMatch crm2pgene:CisRegulatoryModule .

hcrm:Gene rdf:type owl:Class ;
    rdfs:label "Gene" ;
    skos:exactMatch crm2pgene:Gene .

hcrm:TranscriptionFactor rdf:type owl:Class ;
    rdfs:label "Transcription Factor" ;
    skos:exactMatch crm2pgene:TranscriptionFactor .

hcrm:Disease rdf:type owl:Class ;
    rdfs:label "Disease" ;
    skos:exactMatch crm2pgene:Disease .

hcrm:GenomicCoordinate rdf:type owl:Class ;
    rdfs:label "Genomic Coordinate" .

# Properties
hcrm:hasGenomicCoordinate rdf:type owl:ObjectProperty ;
    rdfs:label "has genomic coordinate" ;
    rdfs:domain hcrm:CisRegulatoryModule ;
    rdfs:range hcrm:GenomicCoordinate .

hcrm:targetsGene rdf:type owl:ObjectProperty ;
    rdfs:label "targets gene" ;
    rdfs:domain hcrm:CisRegulatoryModule ;
    rdfs:range hcrm:Gene ;
    skos:exactMatch crm2pgene:targetsGene .

hcrm:associatedWithDisease rdf:type owl:ObjectProperty ;
    rdfs:label "associated with disease" ;
    rdfs:domain hcrm:CisRegulatoryModule ;
    rdfs:range hcrm:Disease ;
    skos:exactMatch crm2pgene:associatedWithDisease .

hcrm:interactsWithTranscriptionFactor rdf:type owl:ObjectProperty ;
    rdfs:label "interacts with transcription factor" ;
    rdfs:domain hcrm:CisRegulatoryModule ;
    rdfs:range hcrm:TranscriptionFactor ;
    skos:exactMatch crm2pgene:interactsWithTranscriptionFactor .

hcrm:hasExperimentalEvidence rdf:type owl:DatatypeProperty ;
    rdfs:label "has experimental evidence" ;
    rdfs:domain hcrm:CisRegulatoryModule ;
    rdfs:range xsd:string .

hcrm:hasBiosampleName rdf:type owl:DatatypeProperty ;
    rdfs:label "has biosample name" ;
    rdfs:domain hcrm:CisRegulatoryModule ;
    rdfs:range xsd:string .

hcrm:hasScore rdf:type owl:DatatypeProperty ;
    rdfs:label "has score" ;
    rdfs:domain hcrm:CisRegulatoryModule ;
    rdfs:range xsd:float .

hcrm:hasPMID rdf:type owl:DatatypeProperty ;
    rdfs:label "has PMID" ;
    rdfs:domain hcrm:CisRegulatoryModule ;
    rdfs:range xsd:string ;
    skos:exactMatch pubmed:PMID .
