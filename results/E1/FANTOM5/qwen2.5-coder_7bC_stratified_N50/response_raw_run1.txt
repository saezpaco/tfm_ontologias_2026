```turtle
@prefix rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#> .
@prefix rdfs: <http://www.w3.org/2000/01/rdf-schema#> .
@prefix owl: <http://www.w3.org/2002/07/owl#> .
@prefix skos: <http://www.w3.org/2004/02/skos/core#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .
@prefix obo: <http://purl.obolibrary.org/obo/> .

owl:Ontology {
    rdf:type owl:Ontology ;
    rdfs:label "Genomic Enhancer Ontology" ;
    skos:definition "An ontology for representing genomic enhancers and their properties." ;
}

crm:Enhancer a owl:Class ;
    rdfs:subClassOf crm:BiologicalEntity ;
    skos:prefLabel "Enhancer" ;
    skos:definition "A region of DNA that can enhance the transcription of other genes." ;
    skos:exactMatch obo:SO_0001765 .

crm:GenomicCoordinate a owl:Class ;
    rdfs:subClassOf crm:BiologicalEntity ;
    skos:prefLabel "Genomic Coordinate" ;
    skos:definition "A specific location on a chromosome." ;

crm:StartCoordinate a owl:DatatypeProperty ;
    rdfs:domain crm:GenomicCoordinate ;
    rdfs:range xsd:integer ;
    skos:prefLabel "Start Coordinate" ;
    skos:definition "The starting position of the genomic coordinate." .

crm:EndCoordinate a owl:DatatypeProperty ;
    rdfs:domain crm:GenomicCoordinate ;
    rdfs:range xsd:integer ;
    skos:prefLabel "End Coordinate" ;
    skos:definition "The ending position of the genomic coordinate." ;

crm:Evidence a owl:Class ;
    rdfs:subClassOf crm:BiologicalEntity ;
    skos:prefLabel "Evidence" ;
    skos:definition "Information or data supporting the existence or properties of an enhancer." ;

crm:Supports a owl:ObjectProperty ;
    rdfs:domain crm:Evidence ;
    rdfs:range crm:Enhancer ;
    skos:prefLabel "Supports" ;
    skos:definition "Indicates that evidence supports the existence or properties of an enhancer." ;

crm:HasCoordinate a owl:ObjectProperty ;
    rdfs:domain crm:Enhancer ;
    rdfs:range crm:GenomicCoordinate ;
    skos:prefLabel "Has Coordinate" ;
    skos:definition "Indicates that an enhancer has a specific genomic coordinate." ;
}
```
