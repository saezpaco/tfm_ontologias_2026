@prefix rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#> .

@prefix owl: <http://www.w3.org/2002/07/owl#> .

    rdfs:comment "An ontology for dbSUPER database containing information about super-enhancers." ;

    rdfs:subClassOf obo:SIO_010245 ;  # Biological Region

    skos:prefLabel "Start" ;
    skos:definition "The starting position of the genomic coordinate." ;

    rdfs:domain crm:GenomicCoordinate ;
    rdfs:range xsd:integer ;
    skos:prefLabel "Stop" ;
    skos:definition "The ending position of the genomic coordinate." ;

    rdfs:domain crm:Enhancer ;
    rdfs:range xsd:string ;
    skos:prefLabel "Cell Line" ;
    skos:definition "A specific population of cells that can be cultured and propagated." ;

crm:Rank a owl:DatatypeProperty ;
    rdfs:domain crm:Enhancer ;
    rdfs:range xsd:integer ;
    skos:prefLabel "Rank" ;
    skos:definition "The rank or importance of the super-enhancer." ;

    rdfs:range crm:CellLine ;
    skos:prefLabel "Has Cell Line" ;
    skos:definition "A relationship between an enhancer and the cell line it is active in." ;

    rdfs:domain crm:Enhancer ;
    rdfs:range crm:Rank ;
    skos:prefLabel "Has Rank" ;
    skos:definition "A relationship between an enhancer and its rank or importance." ;
