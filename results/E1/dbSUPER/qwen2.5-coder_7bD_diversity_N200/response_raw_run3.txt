```turtle
@prefix rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#> .
@prefix rdfs: <http://www.w3.org/2000/01/rdf-schema#> .
@prefix owl: <http://www.w3.org/2002/07/owl#> .
@prefix skos: <http://www.w3.org/2004/02/skos/core#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .
@prefix obo: <http://purl.obolibrary.org/obo/> .

owl:Ontology {
    rdf:type owl:Ontology ;
    rdfs:label "CRM Enhancer Ontology" ;
    rdfs:comment "An ontology for representing CRM enhancers and their properties." ;
}

crm:Enhancer a owl:Class ;
    skos:prefLabel "CRM Enhancer" ;
    skos:definition "A genomic region that enhances the transcription of a gene or set of genes." ;
    rdfs:subClassOf obo:SO_0001467 ;  # Transcriptional enhancer

crm:GenomicCoordinate a owl:Class ;
    skos:prefLabel "Genomic Coordinate" ;
    skos:definition "A specific location on a chromosome." ;

crm:start a owl:DatatypeProperty , owl:FunctionalProperty ;
    rdfs:domain crm:Enhancer ;
    rdfs:range xsd:integer ;
    skos:prefLabel "Start Position" ;
    skos:definition "The starting position of the enhancer region on the chromosome." ;

crm:end a owl:DatatypeProperty , owl:FunctionalProperty ;
    rdfs:domain crm:Enhancer ;
    rdfs:range xsd:integer ;
    skos:prefLabel "End Position" ;
    skos:definition "The ending position of the enhancer region on the chromosome." ;

crm:evidence a owl:ObjectProperty ;
    rdfs:domain crm:Enhancer ;
    rdfs:range rdf:Resource ;
    skos:prefLabel "Evidence" ;
    skos:definition "Supporting evidence for the existence and function of the enhancer." ;

crm:hasGenomicCoordinate a owl:ObjectProperty ;
    rdfs:domain crm:Enhancer ;
    rdfs:range crm:GenomicCoordinate ;
    skos:prefLabel "Has Genomic Coordinate" ;
    skos:definition "The genomic coordinate associated with the enhancer." ;

crm:evidenceType a owl:DatatypeProperty , owl:FunctionalProperty ;
    rdfs:domain crm:Evidence ;
    rdfs:range xsd:string ;
    skos:prefLabel "Evidence Type" ;
    skos:definition "The type of evidence supporting the enhancer's function." ;

crm:evidenceSource a owl:DatatypeProperty , owl:FunctionalProperty ;
    rdfs:domain crm:Evidence ;
    rdfs:range xsd:string ;
    skos:prefLabel "Evidence Source" ;
    skos:definition "The source of the evidence supporting the enhancer's function." ;

crm:evidenceDescription a owl:DatatypeProperty , owl:FunctionalProperty ;
    rdfs:domain crm:Evidence ;
    rdfs:range xsd:string ;
    skos:prefLabel "Evidence Description" ;
    skos:definition "A description of the evidence supporting the enhancer's function." ;

crm:evidenceConfidence a owl:DatatypeProperty , owl:FunctionalProperty ;
    rdfs:domain crm:Evidence ;
    rdfs:range xsd:string ;
    skos:prefLabel "Evidence Confidence" ;
    skos:definition "The level of confidence in the evidence supporting the enhancer's function." ;

crm:evidenceSupports a owl:ObjectProperty ;
    rdfs:domain crm:Evidence ;
    rdfs:range crm:Enhancer ;
    skos:prefLabel "Supports" ;
    skos:definition "Indicates that the evidence supports the enhancer's function." ;
}
```
