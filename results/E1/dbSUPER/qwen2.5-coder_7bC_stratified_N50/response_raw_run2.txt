```turtle
@prefix rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#> .
@prefix rdfs: <http://www.w3.org/2000/01/rdf-schema#> .
@prefix owl: <http://www.w3.org/2002/07/owl#> .
@prefix skos: <http://www.w3.org/2004/02/skos/core#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .
@prefix obo: <http://purl.obolibrary.org/obo/> .

owl:Ontology {
    rdf:type owl:Ontology ;
    rdfs:label "dbSUPER Ontology" ;
    rdfs:comment "An ontology for cis-regulatory modules (CRM) and enhancers in the human genome." ;
}

crm:Enhancer a owl:Class ;
    skos:prefLabel "Enhancer" ;
    skos:definition "A cis-regulatory module that modulates the transcription rate of target genes." ;
    rdfs:subClassOf obo:SO_0001765 .  # SO term for enhancer

crm:GenomicCoordinates a owl:Class ;
    skos:prefLabel "Genomic Coordinates" ;
    skos:definition "The genomic location of an enhancer." ;

crm:Chromosome a owl:DatatypeProperty ;
    rdfs:domain crm:GenomicCoordinates ;
    rdfs:range xsd:string ;
    skos:prefLabel "Chromosome" ;
    skos:definition "The chromosome on which the enhancer is located." ;
    skos:exactMatch obo:SO_0001217 .  # SO term for chromosome

crm:Start a owl:DatatypeProperty ;
    rdfs:domain crm:GenomicCoordinates ;
    rdfs:range xsd:integer ;
    skos:prefLabel "Start" ;
    skos:definition "The start position of the enhancer on the chromosome." ;
    skos:exactMatch obo:SO_0001218 .  # SO term for start

crm:Stop a owl:DatatypeProperty ;
    rdfs:domain crm:GenomicCoordinates ;
    rdfs:range xsd:integer ;
    skos:prefLabel "Stop" ;
    skos:definition "The stop position of the enhancer on the chromosome." ;
    skos:exactMatch obo:SO_0001219 .  # SO term for stop

crm:Evidence a owl:Class ;
    skos:prefLabel "Evidence" ;
    skos:definition "Experimental evidence supporting the enhancer's function." ;

crm:CellLineActivity a owl:ObjectProperty ;
    rdfs:domain crm:Enhancer ;
    rdfs:range crm:CellLine ;
    skos:prefLabel "Cell Line Activity" ;
    skos:definition "The cell line in which the enhancer is active." ;

crm:TargetGene a owl:ObjectProperty ;
    rdfs:domain crm:Enhancer ;
    rdfs:range obo:GO_0003824 ;  # GO term for gene
    skos:prefLabel "Target Gene" ;
    skos:definition "The gene regulated by the enhancer." ;

crm:Disease a owl:ObjectProperty ;
    rdfs:domain crm:Enhancer ;
    rdfs:range obo:DOID_0050128 ;  # DO term for disease
    skos:prefLabel "Associated Disease" ;
    skos:definition "A disease associated with the enhancer." ;

crm:TranscriptionFactor a owl:ObjectProperty ;
    rdfs:domain crm:Enhancer ;
    rdfs:range obo:GO_0003674 ;  # GO term for transcription factor
    skos:prefLabel "Interacting Transcription Factor" ;
    skos:definition "A transcription factor interacting with the enhancer." ;

crm:CellLine a owl:Class ;
    skos:prefLabel "Cell Line" ;
    skos:definition "A cell line used in experimental studies." ;

crm:Rank a owl:DatatypeProperty ;
    rdfs:domain crm:Enhancer ;
    rdfs:range xsd:integer ;
    skos:prefLabel "Rank" ;
    skos:definition "The rank of the enhancer based on some criteria." ;
}
```
