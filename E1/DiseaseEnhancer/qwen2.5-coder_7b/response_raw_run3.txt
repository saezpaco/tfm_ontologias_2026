```turtle
@prefix rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#> .
@prefix rdfs: <http://www.w3.org/2000/01/rdf-schema#> .
@prefix owl: <http://www.w3.org/2002/07/owl#> .
@prefix skos: <http://www.w3.org/2004/02/skos/core#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .
@prefix obo: <http://purl.obolibrary.org/obo/> .

owl:Ontology {
    rdf:type owl:Ontology ;
    rdfs:label "CRM Enhancer Ontology" ;
    rdfs:comment "A formal representation of CRM enhancers and their properties." ;
}

crm:Enhancer a owl:Class ;
    skos:prefLabel "CRM Enhancer" ;
    skos:definition "A genomic region that enhances the transcriptional activity of a target gene." ;
    obo:IAO_0000115 "http://purl.obolibrary.org/obo/GO_0003723" .

crm:GenomicCoordinate a owl:Class ;
    skos:prefLabel "Genomic Coordinate" ;
    skos:definition "A specific location within the genome." ;
    obo:IAO_0000115 "http://purl.obolibrary.org/obo/SO_0000704" .

crm:StartCoordinate a owl:DatatypeProperty , owl:FunctionalProperty ;
    rdfs:domain crm:GenomicCoordinate ;
    rdfs:range xsd:integer ;
    skos:prefLabel "Start Coordinate" ;
    skos:definition "The starting position of the genomic coordinate." ;
    obo:IAO_0000115 "http://purl.obolibrary.org/obo/SO_0000704" .

crm:EndCoordinate a owl:DatatypeProperty , owl:FunctionalProperty ;
    rdfs:domain crm:GenomicCoordinate ;
    rdfs:range xsd:integer ;
    skos:prefLabel "End Coordinate" ;
    skos:definition "The ending position of the genomic coordinate." ;
    obo:IAO_0000115 "http://purl.obolibrary.org/obo/SO_0000704" .

crm:Evidence a owl:Class ;
    skos:prefLabel "Evidence" ;
    skos:definition "Information supporting the existence or properties of an enhancer." ;
    obo:IAO_0000115 "http://purl.obolibrary.org/obo/SIO_000628" .

crm:EvidenceType a owl:Class ;
    skos:prefLabel "Evidence Type" ;
    skos:definition "The type of evidence supporting an enhancer." ;
    obo:IAO_0000115 "http://purl.obolibrary.org/obo/SIO_000628" .

crm:Supports a owl:ObjectProperty ;
    rdfs:domain crm:Evidence ;
    rdfs:range crm:Enhancer ;
    skos:prefLabel "Supports" ;
    skos:definition "Indicates that the evidence supports the enhancer." ;
    obo:IAO_0000115 "http://purl.obolibrary.org/obo/SIO_000628" .

crm:EvidenceType hasEvidence a owl:ObjectProperty ;
    rdfs:domain crm:EvidenceType ;
    rdfs:range crm:Evidence ;
    skos:prefLabel "Has Evidence" ;
    skos:definition "Indicates that the evidence type is associated with the evidence." ;
    obo:IAO_0000115 "http://purl.obolibrary.org/obo/SIO_000628" .

crm:EvidenceType hasEvidenceType a owl:ObjectProperty ;
    rdfs:domain crm:Evidence ;
    rdfs:range crm:EvidenceType ;
    skos:prefLabel "Has Evidence Type" ;
    skos:definition "Indicates that the evidence is of a certain type." ;
    obo:IAO_0000115 "http://purl.obolibrary.org/obo/SIO_000628" .

crm:Evidence hasGenomicCoordinate a owl:ObjectProperty ;
    rdfs:domain crm:Evidence ;
    rdfs:range crm:GenomicCoordinate ;
    skos:prefLabel "Has Genomic Coordinate" ;
    skos:definition "Indicates that the evidence is associated with a genomic coordinate." ;
    obo:IAO_0000115 "http://purl.obolibrary.org/obo/SIO_000628" .
```
